const PACOTE = "jujutsu-compendium.equipamento-pdn";
const ROTULO = "JLRPG - Equipamentos PdN";

const pack = game.packs.get(PACOTE);
if ( !pack ) {
  ui.notifications.error(`Compêndio "${ROTULO}" (${PACOTE}) não encontrado — crie-o no módulo jujutsu-compendium antes de importar.`);
  return;
}
if ( pack.locked ) await pack.configure({ locked: false });

// ── Pastas por categoria ────────────────────────────────────────────────────
async function criarPasta(nome) {
  const existente = pack.folders.find(f => f.name === nome && !f.folder);
  if ( existente ) return existente;
  const [criada] = await Folder.createDocuments([{ name: nome, type: "Item" }], { pack: pack.collection });
  return criada;
}
const pastas = {};
for ( const categoria of ["Armas Pesadas", "Armas de Projétil", "Armas Leves", "Armas Desarmadas"] )
  pastas[categoria] = await criarPasta(categoria);

// ── Existentes por _id (atualizar em vez de duplicar) ──────────────────────
await pack.getDocuments({});
const existentes = new Map(pack.index.map(i => [i._id, i._id]));
let criados = [], atualizados = [];

for ( const fonte of DADOS.armas ) {
  const dadosItem = { _id: fonte._id, name: fonte.name, type: fonte.type, img: fonte.img,
    system: fonte.system, flags: fonte.flags, folder: pastas[fonte._pasta]?.id ?? null };
  if ( existentes.has(fonte._id) ) {
    await Item.updateDocuments([dadosItem], { pack: pack.collection });
    atualizados.push(fonte.name);
  } else {
    await Item.createDocuments([dadosItem], { pack: pack.collection, keepId: true });
    criados.push(fonte.name);
  }
}

const total = (await pack.getIndex()).size;
const esperado = DADOS.armas.length;
let mensagem = `<strong>Armas de Personagem</strong> — ${total}/${esperado} itens no compêndio.`;
if ( criados.length ) mensagem += `<br>Importadas: ${criados.length} arma(s).`;
if ( atualizados.length ) mensagem += `<br>Atualizadas: ${atualizados.length} arma(s).`;
ChatMessage.create({ content: mensagem, speaker: ChatMessage.getSpeaker() });
ui.notifications.info(`Armas de Personagem: ${total}/${esperado} no compêndio.`);
